const express=require("express"); const cors=require("cors"); const {RtcTokenBuilder,RtcRole}=require("agora-access-token"); const app=express();app.use(cors());app.get("/",(req,res)=>res.send("LIZZY LIVE Audio token server is running 💜")); app.get("/rtc/:channel/:uid",(req,res)=>{ const appId=process.env.AGORA_APP_ID,cert=process.env.AGORA_APP_CERTIFICATE; if(!appId||!cert)return res.status(500).json({error:"Agora credentials missing"}); const channel=req.params.channel,uid=Number(req.params.uid); if(!channel||!Number.isInteger(uid)||uid<=0)return res.status(400).json({error:"Invalid channel or numeric UID"}); const expiry=Math.floor(Date.now()/1000)+3600; const token=RtcTokenBuilder.buildTokenWithUid(appId,cert,channel,uid,RtcRole.PUBLISHER,expiry,expiry); res.json({appId,token,channel,uid}); }); const port=process.env.PORT||3000;app.listen(port,()=>console.log("LIZZY LIVE server listening on "+port));
